22.1-03: the raw EXPLAIN (ANALYZE, BUFFERS) behind S6's index clause
=====================================================================

Recorded for PR #64's review (B3), 2026-10-06, re-measured on the branch at
2a16c21 (migration 000018 in its final, plain CREATE INDEX form; the List
statement printed from handlers.RepositoryListWithCurrentJobSQL, byte-identical
to the one first measured on 652e5b7).

Where: a scratch pgvector/pgvector:pg16 container started for this run, pulled
by the digest pinned in container.go, migrated 1 -> 18 with golang-migrate as
its superuser, then removed. Not port 5434, not compose.

How: psql -v ON_ERROR_STOP=1 -f <the SQL below>, one session: the seed (as the
superuser, with app.current_tenant set per organization), ANALYZE, then the
production List statement as rag_doc_app under organization 1's tenant
(current_user and the tenant asserted in the output), first with the custom
plan and then with plan_cache_mode = force_generic_plan (pgx caches prepared
statements, so production can run either). TIMING OFF and COSTS OFF keep the
output stable; buffers and row counts are measured.

Reading it: the live lookup (SubPlan 1) is a Limit over an Index Scan on
idx_ingestion_jobs_one_live_per_repo; the newest lookup (SubPlan 2, run for
the 50 repositories with no live job) is a Limit over an Index Scan on
idx_ingestion_jobs_repository_recent; neither has a Sort. The plan's one Sort
is the list's own ORDER BY r.created_at, r.id over 100 repository rows, as
before 22.1-03. "actual rows=0 loops=100" on SubPlan 1 is a per-loop average
rounded down: 50 of the 100 loops find a live job.

THE SQL RUN (seed, then the EXPLAIN)
---------------------------------------------------------------------
-- 22.1-03's EXPLAIN fixture, on a scratch database only: one organization,
-- 100 repositories, 99 terminal jobs each spread over 30 days, and a live job
-- on half of them (9,950 jobs), plus a second organization with the same
-- shape so the organization filter has rows to exclude.
DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'rag_doc_app') THEN
    CREATE ROLE rag_doc_app NOSUPERUSER NOBYPASSRLS INHERIT;
  END IF;
END $$;
GRANT USAGE ON SCHEMA public TO rag_doc_app;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO rag_doc_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO rag_doc_app;

CREATE TEMP TABLE seed_orgs (n int, org uuid, project uuid);

DO $$
DECLARE
  o int; org uuid; proj uuid; repo uuid; i int;
BEGIN
  FOR o IN 1..2 LOOP
    INSERT INTO organizations (name, slug) VALUES ('explain-' || o, 'explain-2213-' || o) RETURNING id INTO org;
    INSERT INTO projects (organization_id, name, slug, is_default)
      VALUES (org, 'explain-' || o, 'explain-2213-proj-' || o, true) RETURNING id INTO proj;
    INSERT INTO seed_orgs VALUES (o, org, proj);
    PERFORM set_config('app.current_tenant', org::text, true);
    FOR i IN 1..100 LOOP
      INSERT INTO repositories (project_id, name, git_url)
        VALUES (proj, 'r' || i, 'https://example.test/explain-' || o || '-' || i || '.git')
        RETURNING id INTO repo;
      INSERT INTO ingestion_jobs (organization_id, repository_id, job_type, state, attempts, created_at, updated_at)
        SELECT org, repo, 'full_ingest',
               (ARRAY['completed','completed','completed','dead','superseded'])[1 + (g.k % 5)],
               1, NOW() - (g.k || ' hours')::interval * 7, NOW()
        FROM generate_series(1, 99) AS g(k);
      IF i % 2 = 0 THEN
        INSERT INTO ingestion_jobs (organization_id, repository_id, job_type, state, attempts, created_at)
          VALUES (org, repo, 'full_ingest', 'queued', 0, NOW() - interval '30 days');
      END IF;
    END LOOP;
  END LOOP;
END $$;

ANALYZE ingestion_jobs;
ANALYZE repositories;

SELECT o.n, (SELECT count(*) FROM ingestion_jobs j WHERE j.organization_id = o.org) AS jobs,
       (SELECT count(DISTINCT repository_id) FROM ingestion_jobs j WHERE j.organization_id = o.org) AS repos
FROM seed_orgs o ORDER BY o.n;
SELECT count(*) AS all_jobs FROM ingestion_jobs;

SELECT org AS org1 FROM seed_orgs WHERE n = 1 \gset
BEGIN;
SET LOCAL ROLE rag_doc_app;
SELECT set_config('app.current_tenant', :'org1', true) IS NOT NULL AS tenant_set;
SELECT current_user, current_setting('app.current_tenant') = :'org1' AS tenant_is_org1;
PREPARE list_stmt(uuid, timestamptz, uuid, int) AS
SELECT 
       r.id::text, r.name, r.git_url, r.default_branch, r.github_repo_id,
       r.installation_id::text, r.visibility, r.size_kb, r.archived,
       r.sync_state, r.last_synced_at, r.created_at,
       j.id::text, j.repository_id::text, j.job_type, j.state,
       j.attempts, j.max_attempts, j.run_after,
       j.lease_expires_at,
       (j.state = 'running'
        AND (j.lease_expires_at IS NULL OR j.lease_expires_at < NOW())) AS stalled,
       CASE
         WHEN j.attempts >= j.max_attempts
              AND (j.state = 'queued'
                   OR (j.state = 'running'
                       AND (j.lease_expires_at IS NULL OR j.lease_expires_at < NOW())))
           THEN 'dead_pending'
         WHEN j.state = 'running'
              AND (j.lease_expires_at IS NULL OR j.lease_expires_at < NOW())
           THEN 'stalled'
         WHEN j.state = 'running' THEN 'running'
         WHEN j.state = 'queued'
              AND gi.suspended_at IS NOT NULL AND gi.uninstalled_at IS NULL
           THEN 'deferred_suspended'
         WHEN j.state = 'queued' AND j.attempts > 0 THEN 'retrying'
         WHEN j.state = 'queued' AND j.run_after > NOW() THEN 'scheduled'
         WHEN j.state = 'queued' THEN 'queued'
         ELSE j.state
       END AS status,
       j.last_stage, j.progress, j.needs_rerun, j.last_error,
       j.ingestion_run_id::text,
       j.created_at, j.updated_at
FROM repositories r
LEFT JOIN LATERAL (
  SELECT COALESCE(
    (SELECT j1.id FROM ingestion_jobs j1
      WHERE j1.repository_id = r.id
        AND j1.organization_id = $1
        AND j1.state IN ('queued','running')
      LIMIT 1),
    (SELECT j2.id FROM ingestion_jobs j2
      WHERE j2.repository_id = r.id
        AND j2.organization_id = $1
      ORDER BY j2.created_at DESC, j2.id DESC
      LIMIT 1)
  ) AS job_id
) pick ON true
LEFT JOIN ingestion_jobs j ON j.id = pick.job_id
LEFT JOIN github_installations gi ON gi.id = r.installation_id
WHERE ($2::timestamptz IS NULL OR (r.created_at, r.id) > ($2::timestamptz, $3::uuid))
ORDER BY r.created_at ASC, r.id ASC
LIMIT $4;
\echo === custom plan (first execution) ===
EXPLAIN (ANALYZE, BUFFERS, COSTS OFF, TIMING OFF) EXECUTE list_stmt(:'org1', NULL, NULL, 101);
SET LOCAL plan_cache_mode = force_generic_plan;
\echo === generic plan (plan_cache_mode = force_generic_plan) ===
EXPLAIN (ANALYZE, BUFFERS, COSTS OFF, TIMING OFF) EXECUTE list_stmt(:'org1', NULL, NULL, 101);
ROLLBACK;
