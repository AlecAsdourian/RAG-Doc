    ```go
    import (
        "golang.org/x/oauth2"
        "golang.org/x/oauth2/github"
        "golang.org/x/oauth2/gitlab"
        "os"
    )

    type OAuthConfig struct {
        GitHub *oauth2.Config
        GitLab *oauth2.Config
    }

    func NewOAuthConfig() *OAuthConfig {
        return &OAuthConfig{
            GitHub: &oauth2.Config{
                ClientID:     os.Getenv("GITHUB_CLIENT_ID"),
                ClientSecret: os.Getenv("GITHUB_CLIENT_SECRET"),
                RedirectURL:  os.Getenv("BASE_URL") + "/auth/github/callback",
                Scopes:       []string{"user:email", "read:user"},
                Endpoint:     github.Endpoint,
            },
            GitLab: &oauth2.Config{
                ClientID:     os.Getenv("GITLAB_CLIENT_ID"),
                ClientSecret: os.Getenv("GITLAB_CLIENT_SECRET"),
                RedirectURL:  os.Getenv("BASE_URL") + "/auth/gitlab/callback",
                Scopes:       []string{"read_user", "email"},
                Endpoint:     gitlab.Endpoint,
            },
        }
    }
    ```

    **Create pkg/auth/provisioning.go for user creation:**
    ```go
    package auth

    import (
        "context"
        "fmt"
        "github.com/google/uuid"
        "github.com/jackc/pgx/v5/pgxpool"
    )

    type UserProvisioner struct {
        db *pgxpool.Pool
    }

    func NewUserProvisioner(db *pgxpool.Pool) *UserProvisioner {
        return &UserProvisioner{db: db}
    }

    type User struct {
        ID              uuid.UUID
        SupabaseUserID  uuid.UUID
        Email           string
        FullName        string
    }

    // ProvisionOAuthUser creates or updates user from OAuth provider
    func (p *UserProvisioner) ProvisionOAuthUser(ctx context.Context, provider, email, fullName, providerUserID string) (*User, bool, error) {
        // Check if user exists by email
        var user User
        err := p.db.QueryRow(ctx,
            `SELECT id, supabase_user_id, email, full_name FROM users WHERE email = $1`,
            email,
        ).Scan(&user.ID, &user.SupabaseUserID, &user.Email, &user.FullName)

        if err == nil {
            // User exists, return
            return &user, false, nil
        }

        // User doesn't exist, create new user
        // Generate new UUID for supabase_user_id (temporary - will be real Supabase ID later)
        supabaseUserID := uuid.New()

        err = p.db.QueryRow(ctx,
            `INSERT INTO users (supabase_user_id, email, full_name)
             VALUES ($1, $2, $3)
             RETURNING id, supabase_user_id, email, full_name`,
            supabaseUserID, email, fullName,
        ).Scan(&user.ID, &user.SupabaseUserID, &user.Email, &user.FullName)

        if err != nil {
            return nil, false, fmt.Errorf("failed to create user: %w", err)
        }

        return &user, true, nil
    }

    // CreateOrganizationForUser creates org and adds user as owner
    func (p *UserProvisioner) CreateOrganizationForUser(ctx context.Context, userID uuid.UUID, orgName, orgSlug string) (uuid.UUID, error) {
        var orgID uuid.UUID

        // Create organization
        err := p.db.QueryRow(ctx,
            `INSERT INTO organizations (name, slug)
             VALUES ($1, $2)
             RETURNING id`,
            orgName, orgSlug,
        ).Scan(&orgID)

        if err != nil {
            return uuid.Nil, fmt.Errorf("failed to create organization: %w", err)
        }

        // Add user as owner
        _, err = p.db.Exec(ctx,
            `INSERT INTO organization_memberships (user_id, organization_id, role)
             VALUES ($1, $2, 'owner')`,
            userID, orgID,
        )

        if err != nil {
            return uuid.Nil, fmt.Errorf("failed to add user to organization: %w", err)
        }

        return orgID, nil
    }
    ```

  <decision>Supabase Auth integration approach</decision>
  <context>
    We have two options for integrating our OAuth flow with Supabase Auth:

    **Current state:** OAuth handlers fetch user info, provision to our database, but don't create Supabase Auth session.

    **Need:** Users must have both our database record AND Supabase Auth session for JWT validation to work.
  </context>
  <options>
    <option id="supabase-native">
      <name>Use Supabase Native OAuth (Recommended)</name>
      <pros>
        - Supabase handles entire OAuth flow (redirect, callback, token exchange)
        - Automatic JWT generation with proper claims
        - Built-in session management
        - Less code to maintain
        - Better security (Supabase's OAuth implementation is hardened)
      </pros>
      <cons>
        - Less control over OAuth flow
        - Must use Supabase redirect URLs
        - Our backend becomes passive (listens for webhook/events)
        - Need to implement Supabase webhook handler for user provisioning
      </cons>
    </option>
    <option id="custom-oauth">
      <name>Custom OAuth with Supabase API</name>
      <pros>
        - Full control over OAuth flow
        - Can customize user experience
        - Backend owns entire flow
        - Already partially implemented
      </pros>
      <cons>
        - Must manually create Supabase users via Admin API
        - Must generate custom JWTs with proper claims
        - More code to maintain and secure
        - Potential for bugs in OAuth implementation
      </cons>
    </option>
