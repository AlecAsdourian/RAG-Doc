    **Structure:**
    ```go
    package auth

    import (
        "context"
        "fmt"
        "github.com/lestrrat-go/jwx/v3/jwk"
        "github.com/lestrrat-go/jwx/v3/jwt"
    )

    type JWTValidator struct {
        jwksURL string
        issuer  string
    }

    func NewJWTValidator(config *Config) *JWTValidator {
        return &JWTValidator{
            jwksURL: config.JWKSUrl,
            issuer:  config.SupabaseURL,
        }
    }

    // ValidateToken validates JWT against Supabase JWKS
    func (v *JWTValidator) ValidateToken(ctx context.Context, tokenString string) (jwt.Token, error) {
        // Fetch JWKs from Supabase (cached, auto-refreshed by library)
        set, err := jwk.Fetch(ctx, v.jwksURL)
        if err != nil {
            return nil, fmt.Errorf("failed to fetch JWKs: %w", err)
        }

        // Parse and validate token
        token, err := jwt.Parse(
            []byte(tokenString),
            jwt.WithKeySet(set),           // Use JWKs for verification
            jwt.WithValidate(true),        // Validates exp, nbf, iat
            jwt.WithIssuer(v.issuer),      // Validate issuer claim
        )
        if err != nil {
            return nil, fmt.Errorf("invalid token: %w", err)
        }

        return token, nil
    }

    // ExtractUserID gets user_id from validated token
    func ExtractUserID(token jwt.Token) (string, error) {
        userID := token.Subject()
        if userID == "" {
            return "", fmt.Errorf("missing subject claim")
        }
        return userID, nil
    }

    // ExtractOrganizationID gets organization_id from custom claims
    func ExtractOrganizationID(token jwt.Token) (string, error) {
        claims := token.PrivateClaims()
        orgID, ok := claims["organization_id"].(string)
        if !ok || orgID == "" {
            return "", fmt.Errorf("missing organization_id claim")
        }
        return orgID, nil
    }
    ```

    **Why lestrrat-go/jwx:** Comprehensive JOSE support, JWK auto-refresh, handles algorithm validation, production-tested (from RESEARCH.md).

    **Security notes:**
    - JWK auto-refresh handles key rotation automatically
    - Validates exp (expiration), nbf (not before), iat (issued at) via WithValidate(true)
    - Validates issuer to prevent token substitution attacks
    - Library rejects "none" algorithm by default (JWT security best practice)

    **Custom claims:** organization_id will be added to JWT by Supabase Auth trigger (Plan 3 or future work). For now, error if missing.

  <action>
    Create middleware chain following RESEARCH.md patterns: JWT validation → tenant extraction → RLS context setting.

    **Structure:**
    ```go
    package auth

    import (
        "context"
        "fmt"
        "net/http"
        "strings"
        "github.com/jackc/pgx/v5/pgxpool"
    )

    type contextKey string

    const (
        UserIDKey  contextKey = "user_id"
        OrgIDKey   contextKey = "org_id"
    )

    // JWTAuthMiddleware validates JWT and extracts user_id
    func JWTAuthMiddleware(validator *JWTValidator) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
            return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                // Extract Bearer token from Authorization header
                authHeader := r.Header.Get("Authorization")
                if authHeader == "" {
                    http.Error(w, "Missing authorization header", http.StatusUnauthorized)
                    return
                }

                parts := strings.SplitN(authHeader, " ", 2)
                if len(parts) != 2 || parts[0] != "Bearer" {
                    http.Error(w, "Invalid authorization header format", http.StatusUnauthorized)
                    return
                }

                tokenString := parts[1]

                // Validate JWT
                token, err := validator.ValidateToken(r.Context(), tokenString)
                if err != nil {
                    http.Error(w, "Invalid token", http.StatusUnauthorized)
                    return
                }

                // Extract user_id from subject
                userID, err := ExtractUserID(token)
                if err != nil {
                    http.Error(w, "Invalid token claims", http.StatusUnauthorized)
                    return
                }

                // Add to context
                ctx := context.WithValue(r.Context(), UserIDKey, userID)

                next.ServeHTTP(w, r.WithContext(ctx))
            })
        }
    }

    // TenantMiddleware extracts organization and sets RLS context
    func TenantMiddleware(db *pgxpool.Pool) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
            return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                userID, ok := r.Context().Value(UserIDKey).(string)
                if !ok {
                    http.Error(w, "Missing user context", http.StatusInternalServerError)
                    return
                }

                // Get organization from JWT custom claims
                // (Alternative: query organization_memberships table)
                // For now, require it in JWT
                // TODO: Implement organization selection for multi-org users

                orgID := r.Header.Get("X-Organization-ID")  // Temporary: from header
                if orgID == "" {
                    http.Error(w, "Missing organization context", http.StatusBadRequest)
                    return
                }

                ctx := context.WithValue(r.Context(), OrgIDKey, orgID)

                // Set PostgreSQL RLS context (CRITICAL for tenant isolation)
                conn, err := db.Acquire(ctx)
                if err != nil {
                    http.Error(w, "Database error", http.StatusInternalServerError)
                    return
                }
                defer conn.Release()

                // SET LOCAL ensures variable is cleared at end of transaction
                _, err = conn.Exec(ctx, "SET LOCAL app.current_tenant = $1", orgID)
                if err != nil {
                    http.Error(w, "Failed to set tenant context", http.StatusInternalServerError)
                    return
                }

                next.ServeHTTP(w, r.WithContext(ctx))
            })
        }
    }
    ```

    **Note on organization selection:**
    - For MVP, use X-Organization-ID header (simple but not final solution)
    - Production approach (Plan 3 or later): organization_id in JWT custom claims or query memberships table
    - Multi-org users (belong to multiple orgs) need org selection mechanism

    **Critical RLS pattern:** SET LOCAL per request (from RESEARCH.md). Not SET SESSION (connection pooling would leak context between requests).
