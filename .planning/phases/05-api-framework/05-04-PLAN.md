  <action>
    Add streaming chat endpoint to FastAPI service.

    **Update api/models.py:**
    Add ChatStreamRequest (same as ChatRequest).

    **Update api/routes.py:**
    Add POST /chat/stream endpoint that:
    1. Accepts ChatStreamRequest
    2. Calls AnswerGenerator.generate() (non-streaming first, we'll add true streaming later)
    3. Returns StreamingResponse with event-stream content type
    4. Yields chunks in SSE format: "data: {json}\n\n"

    Initial implementation (full response, chunked delivery):
    ```python
    from fastapi.responses import StreamingResponse
    import json

    @router.post("/chat/stream")
    async def chat_stream(request: ChatRequest):
        async def generate():
            try:
                # Get full answer
                result = answer_generator.generate(
                    query=request.query,
                    repository_id=request.repository_id,
                    top_k=request.top_k,
                )

                # Stream the answer in chunks (simulate streaming)
                answer = result["answer"]
                chunk_size = 20  # characters per chunk

                for i in range(0, len(answer), chunk_size):
                    chunk = answer[i:i+chunk_size]
                    yield f"data: {json.dumps({'type': 'chunk', 'content': chunk})}\n\n"
                    await asyncio.sleep(0.01)  # Small delay between chunks

                # Send final metadata
                yield f"data: {json.dumps({'type': 'done', 'sources': result['sources'], 'query_id': str(result['query_id']), 'cost': result['cost']})}\n\n"

            except Exception as e:
                yield f"data: {json.dumps({'type': 'error', 'error': str(e)})}\n\n"

        return StreamingResponse(
            generate(),
            media_type="text/event-stream",
            headers={
                "Cache-Control": "no-cache",
                "Connection": "keep-alive",
            }
        )
    ```

    Later enhancement (true LLM streaming) can be added when OpenAI streaming is integrated into AnswerGenerator.

    ```go
    type ChatRequest struct {
        Query        string `json:"query"`
        RepositoryID string `json:"repository_id"`
        TopK         int    `json:"top_k,omitempty"`
    }

    type ChatChunk struct {
        Type    string   `json:"type"`    // "chunk", "done", "error"
        Content string   `json:"content,omitempty"`
        Sources []Source `json:"sources,omitempty"`
        QueryID string   `json:"query_id,omitempty"`
        Cost    float64  `json:"cost,omitempty"`
        Error   string   `json:"error,omitempty"`
    }

    type Source struct {
        ChunkID   string `json:"chunk_id"`
        FilePath  string `json:"file_path"`
        StartLine int    `json:"start_line"`
        EndLine   int    `json:"end_line"`
    }
    ```

    **Update pkg/client/rag_client.go:**
    ```go
    func (c *RAGClient) StreamChat(ctx context.Context, req ChatRequest) (<-chan ChatChunk, error) {
        // 1. Create POST request to /chat/stream
        // 2. Set Accept: text/event-stream header
        // 3. Do NOT set timeout on this request (streaming can take minutes)
        // 4. Start goroutine to read SSE stream
        // 5. Return channel that emits ChatChunk

        ch := make(chan ChatChunk)

        go func() {
            defer close(ch)

            // Make request
            body, _ := json.Marshal(req)
            httpReq, _ := http.NewRequestWithContext(ctx, "POST", c.baseURL+"/chat/stream", bytes.NewReader(body))
            httpReq.Header.Set("Content-Type", "application/json")
            httpReq.Header.Set("Accept", "text/event-stream")

            // Use client without timeout for streaming
            streamClient := &http.Client{} // No timeout
            resp, err := streamClient.Do(httpReq)
            if err != nil {
                ch <- ChatChunk{Type: "error", Error: err.Error()}
                return
            }
            defer resp.Body.Close()

            // Read SSE stream line by line
            scanner := bufio.NewScanner(resp.Body)
            for scanner.Scan() {
                select {
                case <-ctx.Done():
                    return // Client disconnected
                default:
                }

                line := scanner.Text()
                if strings.HasPrefix(line, "data: ") {
                    data := strings.TrimPrefix(line, "data: ")
                    var chunk ChatChunk
                    if err := json.Unmarshal([]byte(data), &chunk); err == nil {
                        ch <- chunk
                    }
                }
            }
        }()

        return ch, nil
    }
    ```

    Key points:
    - Use separate http.Client without timeout for streaming
    - Check ctx.Done() in read loop to detect client disconnect
    - Close channel when stream ends

  <name>Task 3: Create SSE chat handler</name>
  <files>services/backend/pkg/api/handlers/chat.go</files>
  <action>
    Create chat handler that forwards SSE stream to frontend.

    **pkg/api/handlers/chat.go:**
    ```go
    package handlers

    type ChatHandler struct {
        ragClient *client.RAGClient
        validate  *validator.Validate
    }

    type ChatRequestBody struct {
        Query        string `json:"query" validate:"required,min=1,max=2000"`
        RepositoryID string `json:"repository_id" validate:"required,uuid"`
        TopK         int    `json:"top_k" validate:"omitempty,min=1,max=20"`
    }

    func (h *ChatHandler) Chat(w http.ResponseWriter, r *http.Request) {
        // 1. Validate request body
        var req ChatRequestBody
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
            render.Render(w, r, ErrInvalidRequest(err))
            return
        }
        if req.TopK == 0 {
            req.TopK = 5
        }
        if err := h.validate.Struct(req); err != nil {
            render.Render(w, r, ErrInvalidRequest(err))
            return
        }

        // 2. Set SSE headers
        w.Header().Set("Content-Type", "text/event-stream")
        w.Header().Set("Cache-Control", "no-cache")
        w.Header().Set("Connection", "keep-alive")
        w.Header().Set("X-Accel-Buffering", "no") // Disable nginx buffering

        flusher, ok := w.(http.Flusher)
        if !ok {
            http.Error(w, "SSE not supported", http.StatusInternalServerError)
            return
        }

        // 3. Start streaming from RAG service
        ctx := r.Context()
        chunks, err := h.ragClient.StreamChat(ctx, client.ChatRequest{
            Query:        req.Query,
            RepositoryID: req.RepositoryID,
            TopK:         req.TopK,
        })
        if err != nil {
            fmt.Fprintf(w, "data: %s\n\n", `{"type":"error","error":"Failed to start chat stream"}`)
            flusher.Flush()
            return
        }

        // 4. Forward chunks to client
        for {
            select {
            case <-ctx.Done():
                // Client disconnected
                return
            case chunk, ok := <-chunks:
                if !ok {
                    // Stream complete
                    return
                }
                data, _ := json.Marshal(chunk)
                fmt.Fprintf(w, "data: %s\n\n", data)
                flusher.Flush()
            }
        }
    }
    ```

    Key SSE patterns:
    - Set correct headers BEFORE writing body
    - Use http.Flusher to push data immediately
    - Check ctx.Done() to detect client disconnect (prevents goroutine leak)
    - X-Accel-Buffering: no for nginx reverse proxy
